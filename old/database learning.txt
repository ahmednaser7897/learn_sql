create table teacher
(
	teacher_ssn int identity(1,1) primary key,
	teacher_Fname nvarchar(10),
	teacher_Minit nvarchar(10),
	teacher_lname nvarchar(10),
)

create table teacher_class
(
--طريقه اضافه الكونسترينت فورين كي اثناء انشاء الجدول
	id int foreign key references class (class_ID) on delete cascade on update cascade,
	ssn int foreign key  references teacher (teacher_ssn)on delete cascade on update cascade,
	no_sup int,
--طريقه عمل برايماري كي مزدوج
	constraint tt_pk primary key (id,ssn)
)

exec sp_help teacher_class;

(
	ssn int foreign key  references teacher (teacher_ssn)on delete cascade on update cascade,

--مثلا اذا كان التايخ الذي نبحث عليه هو "2020-02-02 "نكتب

use pl_project

create table student
(
	std_id int primary key identity(1,1),
	std_fname varchar(50),
	std_lname varchar(50), 
	std_pass int not null,
	std_userName varchar(50)
);

create table course
(
	cou_id int primary key identity(1,1),
	cou_name varchar(50), 
	cou_pass int not null,
	cou_userName varchar(50)
);

alter table student add student_course;
alter table student constraint st_cou foreign key (student_course) references course(cou_id);

create table  instructor
(
	inst_id int primary key identity(1,1),
	inst_name varchar(50), 
	inst_pass int not null,
	inst_userName varchar(50)
);

create table  adminn
(
	admin_id int primary key identity(1,1),
	admin_name varchar(50), 
	admin_pass int not null,
	admin_userName varchar(50)
);

create table  std_reg_cou
(
	student_id int foreign key references student (std_id) on delete cascade on update cascade, 
	course_id int foreign key references course (cou_id) on delete cascade on update cascade, 
	constraint s_r_c primary key(student_id,course_id)	 
);
